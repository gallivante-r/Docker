# Manually running Voting App (need network creation and assignment)

```bash
# docker network create --driver bridge voting-app  (explicit --driver bridge)
docker network create voting-app                  # (implicit --driver bridge)

docker network rm voting-app

docker run -d --name=redis --network voting-app redis
docker run -d --name=db --network voting-app -e POSTGRES_PASSWORD=postgres postgres

docker run -d --name=vote --network voting-app -p 5000:80 vote
docker run -d --name=result --network voting-app -p 5001:80 result
docker run -d --name=worker --network voting-app worker
```

# Docker compose running Voting App (network created by default)

```yaml
services:
  vote:
    image: vote
    ports:
      - "5000:80"

  result:
    image: result
    ports:
      - "8081:80"

  worker:
    image: worker

  redis:
    image: redis:alpine

  db:
    image: postgres:15-alpine
    environment:
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: postgres
    volumes:
      - db-data:/var/lib/postgresql/data

volumes:
  db-data:
```
```bash
# conatiners up
docker-compose -f docker-compose-simple.yml up
# containers down
docker-compose -f docker-compose-simple.yml down
# Create file followed by CTRL + D to exit and save
cat > compose.yaml
# Create file followed by esc->:wq!
vim compose.yaml
```
